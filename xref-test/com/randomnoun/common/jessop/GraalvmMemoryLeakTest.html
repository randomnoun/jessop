13  /**
14   * Test to verify that GraalJS ScriptEngine instances release memory when closed,
15   * and to check whether ScriptEngineManager shares underlying engines.
16   *
17   * <p>Run with: -Xmx512m to make OOM failures visible sooner.
18   */
19  public class GraalvmMemoryLeakTest extends TestCase {
20  
21      /**
22       * Tests that creating and closing GraalJS engines does not leak memory.
23       * If close() properly releases Truffle internals, heap usage should remain stable.
24       * If it leaks, this test will eventually OOM or show unbounded growth.
25       */
26      public void testEngineCloseReleasesMemory() throws ScriptException {
27          long baselineMemory = getUsedMemory();
28          System.out.println("Baseline memory: " + (baselineMemory / 1024 / 1024) + " MB");
29  
30          int iterations = 200;
31          for (int i = 0; i < iterations; i++) {
32              ScriptEngine engine = new ScriptEngineManager().getEngineByName("graal-js");
33              assertNotNull("graal-js engine not found", engine);
34              engine.put("polyglot.js.allowAllAccess", true);
35              engine.eval("var x = {a: 1, b: 'hello', c: [1,2,3]}; x.a + x.c.length;");
36  
37              if (engine instanceof AutoCloseable) {
38                  try {
39                      ((AutoCloseable) engine).close();
40                  } catch (Exception e) {
41                      fail("Failed to close engine: " + e.getMessage());
42                  }
43              } else {
44                  fail("GraalJSScriptEngine does not implement AutoCloseable");
45              }
46  
47              if (i % 50 == 49) {
48                  System.gc();
49                  long currentMemory = getUsedMemory();
50                  System.out.println("After " + (i + 1) + " iterations (with close): " +
51                      (currentMemory / 1024 / 1024) + " MB (delta: +" +
52                      ((currentMemory - baselineMemory) / 1024 / 1024) + " MB)");
53              }
54          }
55  
56          System.gc();
57          long finalMemory = getUsedMemory();
58          long delta = finalMemory - baselineMemory;
59          System.out.println("Final memory delta (with close): " + (delta / 1024 / 1024) + " MB");
60  
61          // If close works properly, delta should be modest (under 200MB for 200 iterations)
62          assertTrue("Memory grew by " + (delta / 1024 / 1024) + " MB after " + iterations +
63              " closed engines - close() may not be releasing Truffle internals",
64              delta < 200 * 1024 * 1024);
65      }

67      /**
68       * Tests that creating engines WITHOUT closing leaks memory.
69       * This is the control test - it should show unbounded growth.
70       */
71      public void testEngineWithoutCloseLeaksMemory() throws ScriptException {
72          long baselineMemory = getUsedMemory();
73          System.out.println("Baseline memory (no-close test): " + (baselineMemory / 1024 / 1024) + " MB");
74  
75          int iterations = 200;
76          for (int i = 0; i < iterations; i++) {
77              ScriptEngine engine = new ScriptEngineManager().getEngineByName("graal-js");
78              assertNotNull("graal-js engine not found", engine);
79              engine.put("polyglot.js.allowAllAccess", true);
80              engine.eval("var x = {a: 1, b: 'hello', c: [1,2,3]}; x.a + x.c.length;");
81              // Deliberately NOT closing the engine
82  
83              if (i % 50 == 49) {
84                  System.gc();
85                  long currentMemory = getUsedMemory();
86                  System.out.println("After " + (i + 1) + " iterations (no close): " +
87                      (currentMemory / 1024 / 1024) + " MB (delta: +" +
88                      ((currentMemory - baselineMemory) / 1024 / 1024) + " MB)");
89              }
90          }
91  
92          System.gc();
93          long finalMemory = getUsedMemory();
94          long delta = finalMemory - baselineMemory;
95          System.out.println("Final memory delta (no close): " + (delta / 1024 / 1024) + " MB");
96  
97          // This test documents the leak - we expect growth without close
98          System.out.println("NOTE: Growth of " + (delta / 1024 / 1024) +
99              " MB confirms that unclosed engines leak memory");
100     }

102     /**
103      * Tests whether ScriptEngineManager returns the same engine instance or new ones.
104      * If it reuses engines, closing one might affect others.
105      */
106     public void testEngineIdentity() {
107         ScriptEngineManager mgr = new ScriptEngineManager();
108         ScriptEngine engine1 = mgr.getEngineByName("graal-js");
109         ScriptEngine engine2 = mgr.getEngineByName("graal-js");
110 
111         System.out.println("engine1: " + System.identityHashCode(engine1) + " (" + engine1.getClass().getName() + ")");
112         System.out.println("engine2: " + System.identityHashCode(engine2) + " (" + engine2.getClass().getName() + ")");
113         System.out.println("Same instance: " + (engine1 == engine2));
114 
115         // Document whether instances are shared
116         if (engine1 == engine2) {
117             System.out.println("WARNING: ScriptEngineManager reuses engine instances - closing one would break others");
118         } else {
119             System.out.println("OK: ScriptEngineManager creates new engine instances each time");
120         }
121     }

123     /**
124      * Tests whether separately-created engines share an underlying polyglot Engine.
125      * Even if ScriptEngine instances differ, they may share a polyglot Engine
126      * which holds the code cache and shape metadata.
127      */
128     public void testSharedPolyglotEngine() throws Exception {
129         ScriptEngineManager mgr = new ScriptEngineManager();
130         ScriptEngine engine1 = mgr.getEngineByName("graal-js");
131         ScriptEngine engine2 = mgr.getEngineByName("graal-js");
132 
133         // Use reflection to check if they share the same polyglot Engine
134         try {
135             java.lang.reflect.Method getPolyglotContext = engine1.getClass().getMethod("getPolyglotContext");
136             Object ctx1 = getPolyglotContext.invoke(engine1);
137             Object ctx2 = getPolyglotContext.invoke(engine2);
138 
139             java.lang.reflect.Method getEngine = ctx1.getClass().getMethod("getEngine");
140             Object polyEngine1 = getEngine.invoke(ctx1);
141             Object polyEngine2 = getEngine.invoke(ctx2);
142 
143             System.out.println("Polyglot Engine 1: " + System.identityHashCode(polyEngine1));
144             System.out.println("Polyglot Engine 2: " + System.identityHashCode(polyEngine2));
145             System.out.println("Shared polyglot Engine: " + (polyEngine1 == polyEngine2));
146 
147             if (polyEngine1 == polyEngine2) {
148                 System.out.println("IMPORTANT: Engines share a polyglot Engine - " +
149                     "compiled code and shapes accumulate in the shared Engine even after Context.close()");
150             } else {
151                 System.out.println("OK: Each ScriptEngine has its own polyglot Engine - " +
152                     "close() should fully release resources");
153             }
154         } catch (NoSuchMethodException e) {
155             System.out.println("Could not reflectively access polyglot context - " + e.getMessage());
156         }
157     }

159     private long getUsedMemory() {
160         Runtime rt = Runtime.getRuntime();
161         return rt.totalMemory() - rt.freeMemory();
162     }
